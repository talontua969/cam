if type(shared._barrageRequest) ~= "function" then
	local _requestFns = {}

	local function _addRequest(fn)
		if type(fn) ~= "function" then
			return
		end

		for _, existing in ipairs(_requestFns) do
			if existing == fn then
				return
			end
		end

		_requestFns[#_requestFns + 1] = fn
	end

	_addRequest(type(syn) == "table" and syn.request)
	_addRequest(type(http) == "table" and http.request)
	_addRequest(type(fluxus) == "table" and fluxus.request)
	_addRequest(request)
	_addRequest(http_request)

	local _requestPrimary

	-- Executors may rewrite the table they are given (Xeno base64-encodes Body in place),
	-- so each call gets a fresh copy.
	local function _copyOptions(options)
		local copy = {}

		for key, value in next, options do
			copy[key] = value
		end

		if type(options.Headers) == "table" then
			local headers = {}

			for key, value in next, options.Headers do
				headers[key] = value
			end

			copy.Headers = headers
		end

		return copy
	end
	local function _normalize(response)
			-- Some executors (Real) return the error text as a bare string instead of a response.
			if type(response) ~= "table" then
				return nil
			end

			local body = response.Body or response.body or response.ResponseBody or response.responseBody or ""
			local status = tonumber(response.StatusCode or response.Status or response.status_code or response.statusCode)
			local failed = response.Success == false or response.success == false
			local succeeded = response.Success == true or response.success == true

			if not status or status == 0 and succeeded then
				status = failed and 0 or 200
			end

			return {
				Body = type(body) == "string" and body or tostring(body or ""),
				Headers = response.Headers or response.headers or {},
				StatusCode = status,
				StatusMessage = response.StatusMessage or response.statusMessage or response.Message or response.message or "",
				Success = status >= 200 and status < 300,
			}
	end

	shared._barrageRequest = function(options)
		if type(options) ~= "table" then
			return nil, "invalid options"
		end

		local lastError = "request unavailable"

		if _requestPrimary then
			local ok, response = pcall(_requestPrimary, _copyOptions(options))
			local normalized = ok and _normalize(response)

			if normalized then
				return normalized
			end

			lastError = ok and "invalid response" or tostring(response)
			_requestPrimary = nil
		end

		for _, fn in ipairs(_requestFns) do
			local ok, response = pcall(fn, _copyOptions(options))
			local normalized = ok and _normalize(response)

			if normalized then
				_requestPrimary = fn

				return normalized
			end

			lastError = ok and "invalid response" or tostring(response)
		end

		return nil, lastError
	end
end
end


local function geniter(f, s, c)
	if type(f) ~= "function" then
		local mt = getmetatable(f)

		if type(mt) == "table" and mt.__iter then
			f, s, c = mt.__iter(f)
		elseif type(f) == "table" then
			f, s, c = next, f, nil
		end
	end

	return function()
		local r = table.pack(f(s, c))

		if r[1] == nil then
			return false
		end

		c = r[1]

		return true, table.unpack(r, 1, r.n)
	end
end

t1.Fish.wants = function(p277)
	if p277 == "steer" then

		return f.AutoFish == true and f.SkipThrow == true
	end

	if p277 == "dive" then
		return f.AutoFish == true and f.FastDive == true and t1.num(f.DiveRate) > 100

t1.Fish.record = function()
	local rec = t1.Fish.rec

	if rec then
		local v656 = t1.num(rec.ChargeStart)

		if (t1.Fish.idleAt or 0) - 0.05 <= v656 then
			return rec
		end
	end

	local now = os.clock()

	if now < (t1.Fish.recAt or 0) then
		return nil
	end

	t1.Fish.recAt = now + 0.25
	t1.Fish.rec = nil

	local v658 = t1.G.fish()
	local getupvalues_ = type(debug) == "table" and debug.getupvalues or getupvalues

	if not v658 or type(v658.QA) ~= "function" or type(getupvalues_) ~= "function" then
		return nil
	end

	local ok, result = pcall(getupvalues_, v658.QA)

	t1.restoreIdentity()

	if not ok or type(result) ~= "table" then
		return nil
	end

	for i = 1, 32 do
		local v663 = result[i]

		if type(v663) == "table" and type(v663.ChargeStart) == "number" and type(v663.Instances) == "table" then
			t1.Fish.rec = v663

			return v663
		end
	end

	return nil
end

t1.Fish.warp = function(p278)
	local now = os.clock()
	local warpAt = t1.Fish.warpAt

	t1.Fish.warpAt = now

	if not t1.Fish.wants("throw") and not t1.Fish.wants("dive") then
		return
	end

	local v666 = t1.Fish.record()

	if not v666 then
		return
	end

	if p278 ~= "Throwing" or not t1.Fish.wants("throw") then
		if p278 == "Underwater" and warpAt then
			if type(v666.RunStart) == "number" then
				local v667 = t1.Fish.diveRate()

				if v667 > 1 then
					v666.RunStart -= (v667 - 1) * math.min(0.1, now - warpAt)
				end
			end
		end
	elseif v666.Phase == "Flight" and type(v666.FlightTime) == "number" then
		v666.FlightStart = now - v666.FlightTime
	elseif v666.Phase == "Impact" then
		local GameConfig = t1.G.cfg("GameConfig")

		v666.ImpactAt = now - (GameConfig and type(GameConfig.Fishing) == "table" and tonumber(GameConfig.Fishing.SplashTime) or 0.35) - 0.01
	end
end

		local state = t1.Fish.state

		if state == "Throwing" or state == "Underwater" then
			t1.Fish.warp(state)
		end

		if state == "Underwater" and t1.Fish.wants("steer") then
			t1.Gates.step()
		elseif state == "Reeling" and t1.Fish.wants("reel") then
			local now = os.clock()

			if now < t1.Fish.nextTap then
				return
			end

			if t1.G.phase() ~= "Reeling" then
				return
			end

			local GameConfig = t1.G.cfg("GameConfig")
			local Reel = GameConfig and type(GameConfig.Reel) == "table" and GameConfig.Reel or {}
			local AutoFish = f.AutoFish

			if AutoFish then
				AutoFish = t1.K.reelRate(f.ReelRule, Reel.MaxTapRate, (t1.G.cfg("CatchResistanceConfig") or {}).ReferenceTapRate, math.random())
			end

			AutoFish = AutoFish or f.ReelRate

			local v674 = t1.K.tapGap(AutoFish, Reel.MaxTapRate)

			t1.Fish.nextTap = math.max(t1.Fish.nextTap + v674, now - v674)

			local castId = t1.Fish.castId

			if castId ~= nil and t1.Fish.burst ~= castId and t1.num(Reel.MaxTapRate) <= AutoFish then
				t1.Fish.burst = castId

				for i = 1, math.max(0, t1.num(Reel.TapBurst) - 1) do
					t1.G.fire("FishingAction", "Tap", castId)
				end
			end

			t1.G.tap(castId)
		elseif t1.Gates.steering then
			t1.Gates.unsteer()
		end
	end
end

	if p302 == "No Remote" then
		t1.status(p301, "[UNAVAILABLE] " .. p303 .. " | No Remote")
		t1.backoff(p301, 300, v751, p305)

		return
	end

	local backoff, v754, v755

	if v752:find("slow down") then
		p304 = math.min(p304 or 5, 1)
		t1.status(p301, "[RETRYING] " .. p303 .. " | " .. v751)
		backoff = t1.backoff
		v754 = p301
		v755 = p304 or 5
	elseif v752:find("enough") or v752:find("afford") or v752:find("need more") then
		local v756 = p305 or p301
		local now = os.clock()
		local v758 = t1.Short[v756]

		if not v758 or now - v758.at > 300 then
			v758 = {
				n = 0
			}
		end

		v758.n = math.min(v758.n + 1, 5)
		v758.at = now
		t1.Short[v756] = v758
		v755 = math.max(p304 or 5, 15 * 2 ^ (v758.n - 1))
		t1.status(p301, "[RETRYING] " .. p303 .. " | " .. v751)
		backoff = t1.backoff
		v754 = p301
		v755 = v755 or 5
	else
		t1.status(p301, "[RETRYING] " .. p303 .. " | " .. v751)
		backoff = t1.backoff
		v754 = p301
		v755 = p304 or 5
	end
